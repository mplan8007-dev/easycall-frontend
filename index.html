// --- 템플릿 다운로드 (이름과 동은 선택 사항임을 명시) --- function downloadCsvTemplate() { const csvTemplate = "voter_name,phone,dong\n홍길동,010-1234-5678,대천1동\n,010-9876-5432,\n박민수,010-5555-4444,오천면"; const blob = new Blob(["\uFEFF" + csvTemplate], { type: 'text/csv;charset=utf-8;' }); const url = URL.createObjectURL(blob); const link = document.createElement("a"); link.setAttribute("href", url); link.setAttribute("download", "이지콜_유권자명부_표준양식.csv"); document.body.appendChild(link); link.click(); document.body.removeChild(link); } // --- CSV 명부 검증 (오직 'phone' 컬럼만 필수 검증) --- function handleCsvUpload(event) { const file = event.target.files[0]; if (!file) return; const reader = new FileReader(); reader.onload = function(e) { const text = e.target.result; const lines = text.split('\n'); if (lines.length === 0) { alert("빈 파일입니다."); return; } // 헤더 검증 (phone 컬럼만 필수 존재 확인, voter_name과 dong은 없어도 통과) const headers = lines[0].trim().split(','); if (!headers.includes('phone')) { alert("🚨 [양식 오류]\n업로드하신 CSV 파일에 필수 컬럼인 'phone'(전화번호)이 누락되어 있습니다. 올바른 양식으로 다시 업로드해 주세요."); document.getElementById('csv-file-input').value = ''; return; } let validCount = 0; for (let i = 1; i < lines.length; i++) { if (lines[i].trim()) validCount++; } document.getElementById('upload-status').innerText = `✓ 검증 완료! (이름·동 선택 입력 허용) 총 ${validCount}건의 명부가 안전하게 큐에 적재되었습니다.`; }; reader.readAsText(file, 'utf-8'); }
